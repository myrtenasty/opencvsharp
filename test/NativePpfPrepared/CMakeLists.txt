cmake_minimum_required(VERSION 3.20)
project(NativePpfPrepared LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OpenCV_STATIC ON)
find_package(OpenCV REQUIRED COMPONENTS core flann surface_matching)
if(MSVC)
  # The local baseline build exports Unix pthread in static link interfaces.
  foreach(t IN ITEMS opencv_core opencv_flann opencv_surface_matching)
    foreach(p IN ITEMS INTERFACE_LINK_LIBRARIES IMPORTED_LINK_INTERFACE_LIBRARIES_RELEASE)
      get_target_property(libs ${t} ${p})
      if(libs)
        list(FILTER libs EXCLUDE REGEX "pthread")
        set_property(TARGET ${t} PROPERTY ${p} "${libs}")
      endif()
    endforeach()
  endforeach()
endif()
add_executable(ppf_prepared prepared.cpp)
target_include_directories(ppf_prepared PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ppf_prepared PRIVATE ${OpenCV_LIBS})
enable_testing()
add_test(NAME ppf_prepared COMMAND ppf_prepared)
